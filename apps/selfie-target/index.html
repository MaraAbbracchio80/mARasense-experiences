import * as ecs from '@8thwall/ecs' ecs.registerComponent({ name: 'selfie-target-controller', add: (world, component) => { console.log('selfie-target-controller avviato') const selfieBtn = document.getElementById('selfieBtn') const hint = document.getElementById('hint') if (!selfieBtn) { console.warn('selfieBtn non trovato nell HTML') return } selfieBtn.style.display = 'none' selfieBtn.addEventListener('click', () => { window.location.href = '../selfie/index.html?player=dilorenzo' }) world.events.addListener( world.events.globalId, ecs.events.REALITY_IMAGE_FOUND, (event) => { console.log('REALITY_IMAGE_FOUND:', event) const targetName = event?.data?.name if (targetName === 'target22') { selfieBtn.style.display = 'block' if (hint) { hint.innerHTML = 'Maglia riconosciuta. Ora puoi scattare il selfie.' } } } ) world.events.addListener( world.events.globalId, ecs.events.REALITY_IMAGE_LOST, (event) => { console.log('REALITY_IMAGE_LOST:', event) const targetName = event?.data?.name if (targetName === 'target22') { selfieBtn.style.display = 'none' if (hint) { hint.innerHTML = 'Inquadra la maglia di Di Lorenzo per attivare il selfie.' } } } ) }, })
